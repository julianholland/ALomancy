def generate_kpts(
    cell: np.ndarray, periodic_3d: bool = True, kspacing: float = 0.1
) -> np.ndarray:
    cell_lengths = np.linalg.norm(cell, axis=1)
    kpts = np.ceil(2 * np.pi / (cell_lengths * kspacing)).astype(int)
    return kpts if periodic_3d else np.array([kpts[0], kpts[1], 1])

def kpoint_grid(atoms: Atoms, k_spacing: float) -> list[int]:
    """The k-point grid for *atoms* at *k_spacing* (1/Angstrom, 2 pi
    included, as VASP's KSPACING): ceil(2 pi / (|a_i| k_spacing)) along each
    periodic axis (generate_kpts' formula), 1 along each non-periodic one.
    Call it on the structure's own pbc, before any vacuum padding: a slab
    (pbc T T F) gets n x m x 1, a molecule or isolated atom Gamma only,
    instead of k-points along the vacuum."""
    if k_spacing is None or k_spacing <= 0:
        raise ValueError(f"k_spacing must be a positive number, got {k_spacing!r}.")
    lengths = np.linalg.norm(np.asarray(atoms.cell), axis=1)
    grid = []
    for periodic, length in zip(atoms.pbc, lengths, strict=True):
        if periodic and length > 0:
            grid.append(max(1, int(np.ceil(2 * np.pi / (length * k_spacing)))))
        else:
            grid.append(1)
    return grid




_MEMORY_UNITS_MB = {
    "": 1,  # a bare number is MB, as Slurm reads --mem
    "K": 1 / 1024,
    "KB": 1 / 1024,
    "M": 1,
    "MB": 1,
    "G": 1024,
    "GB": 1024,
    "T": 1024 * 1024,
    "TB": 1024 * 1024,
}

def parse_memory_mb(value: object) -> int:
    """A memory amount as written in an HPC profile's node_info
    (``"120GB"``, ``"90G"``, ``"512M"``, or a bare number of MB, Slurm's
    reading of ``--mem``) in MB. Raises ValueError for anything else."""
    if isinstance(value, int | float) and not isinstance(value, bool) and value > 0:
        return int(value)
    if isinstance(value, str):
        match = re.fullmatch(r"\s*([0-9]*\.?[0-9]+)\s*([A-Za-z]*)\s*", value)
        if match and match.group(2).upper() in _MEMORY_UNITS_MB:
            mb = float(match.group(1)) * _MEMORY_UNITS_MB[match.group(2).upper()]
            if mb > 0:
                return int(mb)
    raise ValueError(
        f"Can't read memory amount {value!r}: expected e.g. '120GB', '90G', "
        "'512M' or a number of MB."
    )

# %maxcore, and its manual recommends about 75% of the physical memory.
_DEFAULT_MEMORY_FRACTION = 0.75

class ParallelResources:
    """What one evaluator job may use, from the HPC profile's node_info and
    the user's overrides (resolve_parallel_resources). Backends encode it
    their own way: in the launch command (QE/VASP through srun) or in the
    input file (ORCA %pal/%maxcore, Gaussian %nprocshared/%mem)."""

    nprocs: int
    mem_per_proc_mb: int
    threads_per_proc: int

def resolve_parallel_resources(
    node_info: dict,
    *,
    nprocs: int | None = None,
    mem_per_proc_mb: int | None = None,
    where: str = "high_accuracy_evaluation",
) -> ParallelResources:
    """The processes and per-process memory for one evaluator job.

    Defaults: ``node_info.ranks_per_system`` processes, each with
    _DEFAULT_MEMORY_FRACTION of the node's memory divided between them.
    A user-set *nprocs* or *mem_per_proc_mb* wins over the profile, but
    raises ValueError when the job can't hold it: every evaluator job runs
    on one node (configs/remote_info.get_remote_info), so at most
    ``ranks_per_node * threads_per_rank`` processes, and all of them
    together within ``max_mem_per_node``. *where* names the config block
    in the messages."""
    for key in ("ranks_per_system", "ranks_per_node"):
        if not _is_positive_int(node_info.get(key)):
            raise ValueError(
                f"The HPC profile's node_info.{key} must be a positive integer "
                f"(needed for {where}), got {node_info.get(key)!r}."
            )
    threads = node_info.get("threads_per_rank", 1)
    if not _is_positive_int(threads):
        raise ValueError(
            "The HPC profile's node_info.threads_per_rank must be a positive "
            f"integer, got {threads!r}."
        )
    if "max_mem_per_node" not in node_info:
        raise ValueError(
            f"The HPC profile's node_info.max_mem_per_node is required for "
            f"{where} (the per-process memory is derived from it)."
        )
    node_mb = parse_memory_mb(node_info["max_mem_per_node"])
    cores = int(node_info["ranks_per_node"]) * int(threads)

    if nprocs is None:
        nprocs = int(node_info["ranks_per_system"])
    elif not _is_positive_int(nprocs):
        raise ValueError(f"{where}: nprocs must be a positive integer, got {nprocs!r}.")
    elif nprocs > cores:
        raise ValueError(
            f"{where}: nprocs {nprocs} doesn't fit the job: it runs on one node "
            f"with {cores} cores (node_info.ranks_per_node "
            f"{node_info['ranks_per_node']} x threads_per_rank {threads}). "
            f"Lower nprocs to at most {cores}, or use an HPC profile with "
            "larger nodes."
        )

    if mem_per_proc_mb is None:
        mem_per_proc_mb = int(_DEFAULT_MEMORY_FRACTION * node_mb / nprocs)
        if mem_per_proc_mb < 1:
            raise ValueError(
                f"{where}: node_info.max_mem_per_node "
                f"{node_info['max_mem_per_node']!r} leaves under 1 MB for each "
                f"of {nprocs} processes."
            )
    elif not _is_positive_int(mem_per_proc_mb):
        raise ValueError(
            f"{where}: the memory per process must be a positive integer "
            f"(MB), got {mem_per_proc_mb!r}."
        )
    elif nprocs * mem_per_proc_mb > node_mb:
        raise ValueError(
            f"{where}: {nprocs} processes x {mem_per_proc_mb} MB = "
            f"{nprocs * mem_per_proc_mb} MB is more than the node's "
            f"{node_mb} MB (node_info.max_mem_per_node "
            f"{node_info['max_mem_per_node']!r}). Lower the memory per process "
            f"to at most {node_mb // nprocs} MB (about "
            f"{int(_DEFAULT_MEMORY_FRACTION * node_mb / nprocs)} MB leaves "
            "headroom), or use fewer processes."
        )
    return ParallelResources(
        nprocs=int(nprocs),
        mem_per_proc_mb=int(mem_per_proc_mb),
        threads_per_proc=int(threads),
    )

def pad_non_periodic_axes(atoms: Atoms, vacuum: float = 10.0) -> None:
    """Make *atoms* fully periodic, in place, for a code with no mixed
    boundary conditions (VASP, FHI-aims). Only the non-periodic axes are
    resized, to the structure's extent plus *vacuum* on each side; a
    slab's periodic in-plane cell is left as it is. A structure with no
    cell on a non-periodic axis gets one along it. No-op when already
    fully periodic."""
    if atoms.pbc.all():
        return
    open_axes = [i for i in range(3) if not atoms.pbc[i]]
    atoms.center(vacuum=vacuum, axis=open_axes)
    atoms.pbc = True




def _build_srun_command(para_info_dict: dict, executable_and_flags: str) -> str:
    # --mem=0 means "use all memory already granted to the job" (documented
    # Slurm sentinel), not "request 0 memory". This step runs nested inside
    # a job whose own #SBATCH header may already set --mem; if it were, the
    # enclosing batch script's process itself is accounted as the job's
    # first step and, on some Slurm configs, is credited with the *entire*
    # job memory allocation. A nested srun step that then asks for its own
    # explicit sub-amount (e.g. --mem=60GB) competes with that reservation
    # and fails immediately with "Unable to create step ... Memory required
    # by task is not available", regardless of how large the job's total
    # --mem is. --mem=0 avoids the conflict by inheriting rather than
    # re-requesting.
    return (
        f"srun --ntasks={para_info_dict['ranks_per_system']} "
        f"--tasks-per-node={para_info_dict['ranks_per_node']} "
        f"--cpus-per-task={para_info_dict['threads_per_rank']} "
        f"--distribution=block:block "
        f"--hint=nomultithread "
        f"--mem=0 "
        f"{executable_and_flags}"
    )

def refresh_dft_labels(atoms: Atoms, source: str = "DFT") -> Atoms:
    """Replace inherited labels from the current, explicitly DFT calculator.

    Never call this on a generated candidate with a surrogate calculator.
    Keep ASE's energy convention; store free_energy separately when available.
    """
    if atoms.calc is None:
        raise ValueError("Cannot refresh DFT labels without a calculator")
    if getattr(atoms.calc, "converged", None) is False:
        raise ValueError("DFT electronic calculation did not converge")
    clean = clean_structures([atoms], "DFT", label_source="calculator")[0]
    atoms.info["REF_energy"] = clean.info["REF_energy"]
    atoms.set_array("REF_forces", clean.arrays["REF_forces"].copy())
    # Stress too: replace or drop whatever the structure carried in (an MD
    # frame inherits its seed's info), so it never outlives its source.
    if "REF_stresses" in clean.info:
        atoms.info["REF_stresses"] = clean.info["REF_stresses"]
    else:
        atoms.info.pop("REF_stresses", None)
    atoms.info["REF_label_source"] = source
    if getattr(atoms.calc, "converged", None) is True:
        atoms.info["dft_converged"] = True
    atoms.info["REF_energy_convention"] = "energy"
    free_energy = atoms.calc.results.get("free_energy")
    if free_energy is not None and np.isfinite(free_energy):
        atoms.info["REF_free_energy"] = float(free_energy)
    else:
        atoms.info.pop("REF_free_energy", None)
    return atoms




def _write_dft_result(atoms: Atoms, out_dir: str, name: str) -> None:
    write(Path(out_dir, f"{name}.xyz"), atoms, format="extxyz")
    logger.debug("Writing structures to %s as %s.xyz", out_dir, name)


INITIAL_FRAME_SUFFIX = "_initial"
_SNAPSHOT_PROPERTIES = ("energy", "free_energy", "forces", "stress")
_RUN_STATISTICS = (
    "geometry_converged",
    "geometry_steps",
    "geometry_max_steps",
    "geometry_fmax_target",
    "dft_wall_time_s",
)


def _snapshot(atoms: Atoms) -> Atoms:
    """A copy of *atoms* holding the DFT results already computed for its
    current geometry, read from calc.results only: asking the live
    calculator for a property it didn't compute (e.g. stress with QE
    tstress off) would rerun the whole calculation."""
    results = atoms.calc.results
    copy: Atoms = atoms.copy()
    copy.calc = SinglePointCalculator(
        copy, **{k: results[k] for k in _SNAPSHOT_PROPERTIES if k in results}
    )
    for key in _RUN_STATISTICS:
        copy.info.pop(key, None)
    return copy


def _run_sp(
    input_structure: Atoms,
    out_dir: str,
    job_dict: dict,
    create_calc_fn: Callable,
) -> Atoms:
    Path(out_dir).mkdir(exist_ok=True, parents=True)
    input_structure.calc = create_calc_fn(input_structure, job_dict, out_dir)
    start = time.perf_counter()
    input_structure.get_potential_energy()
    wall_time = time.perf_counter() - start
    refresh_dft_labels(input_structure, str(Path(out_dir).resolve()))
    # Per-structure run statistics for the loop report (analysis/report).
    input_structure.info["dft_wall_time_s"] = wall_time
    _write_dft_result(input_structure, out_dir, job_dict["name"])
    return input_structure


def _run_go(
    input_structure: Atoms,
    out_dir: str,
    job_dict: dict,
    create_calc_fn: Callable,
    opt_prefix: str = "opt",
) -> Atoms:
    Path(out_dir).mkdir(exist_ok=True, parents=True)
    input_structure.calc = create_calc_fn(input_structure, job_dict, out_dir)
    opt = BFGS(
        input_structure,
        logfile=str(Path(out_dir, f"{opt_prefix}.log")),
        trajectory=str(Path(out_dir, f"{opt_prefix}.traj")),
    )
    fmax = job_dict.get("fmax", 0.05)
    steps = job_dict.get("max_num_of_relax_steps", 200)
    start = time.perf_counter()
    # Step 0 is a full DFT calculation of the unrelaxed geometry; BFGS's
    # first step reuses this cached result, so keeping it costs nothing.
    input_structure.get_forces()
    initial = _snapshot(input_structure)
    dft_converged = getattr(input_structure.calc, "converged", None)
    converged = opt.run(fmax=fmax, steps=steps)
    wall_time = time.perf_counter() - start
    if not converged:
        logger.warning(
            "Geometry optimization in %s did not reach fmax=%.4g eV/Angstrom "
            "within %d steps; keeping the best structure found rather than "
            "discarding the completed DFT computation.",
            out_dir,
            fmax,
            steps,
        )
    refresh_dft_labels(input_structure, str(Path(out_dir).resolve()))
    input_structure.info["geometry_converged"] = bool(converged)
    # Per-structure run statistics for the loop report (analysis/report).
    input_structure.info["geometry_steps"] = int(opt.nsteps)
    input_structure.info["geometry_max_steps"] = int(steps)
    input_structure.info["geometry_fmax_target"] = float(fmax)
    input_structure.info["dft_wall_time_s"] = wall_time
    input_structure.info["geometry_step"] = int(opt.nsteps)
    _write_dft_result(input_structure, out_dir, job_dict["name"])
    if opt.nsteps > 0 and dft_converged is not False:
        # The starting geometry is training data too (filters decide).
        refresh_dft_labels(initial, str(Path(out_dir).resolve()))
        if dft_converged:
            initial.info["dft_converged"] = True
        initial.info["geometry_step"] = 0
        initial.info["geometry_initial"] = True
        _write_dft_result(initial, out_dir, job_dict["name"] + INITIAL_FRAME_SUFFIX)
    return input_structure
